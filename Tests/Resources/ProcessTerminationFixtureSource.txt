// Original, deliberately inert fixture. Tests own and reap every invocation.
#include <signal.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>
static volatile sig_atomic_t reexec_requested = 0;
static void request_reexec(int signal_number) { (void)signal_number; reexec_requested = 1; }
int main(int argc, char **argv) {
    if (argc != 3) return 2;
    // Process launch inherits the calling thread's signal mask. An inherited
    // blocked SIGALRM must not defeat the fixture's independent lifetime bound.
    sigset_t lifetime_signals;
    sigemptyset(&lifetime_signals);
    sigaddset(&lifetime_signals, SIGALRM);
    sigaddset(&lifetime_signals, SIGTERM);
    sigaddset(&lifetime_signals, SIGUSR1);
    if (sigprocmask(SIG_UNBLOCK, &lifetime_signals, NULL) != 0) return 8;
    signal(SIGALRM, SIG_DFL);
    alarm(30); // bounded fixture lifetime, including failed test setup/cleanup
    signal(SIGTERM, strcmp(argv[1], "ignore-term") == 0 ? SIG_IGN : SIG_DFL);
    signal(SIGUSR1, request_reexec);
    sigset_t blocked, previous;
    sigemptyset(&blocked);
    sigaddset(&blocked, SIGUSR1);
    if (sigprocmask(SIG_BLOCK, &blocked, &previous) != 0) return 6;
    int ready = open(argv[2], O_WRONLY | O_CREAT | O_TRUNC, 0600);
    if (ready < 0) return 3;
    if (write(ready, "ready", 5) != 5) return 4;
    close(ready);
    while (!reexec_requested) sigsuspend(&previous);
    if (sigprocmask(SIG_SETMASK, &previous, NULL) != 0) return 7;
    execv(argv[0], argv);
    return 5;
}
